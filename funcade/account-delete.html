package com.bwm.funcade.account; import jakarta.validation.Valid; import jakarta.validation.constraints.Email; import jakarta.validation.constraints.NotBlank; import jakarta.validation.constraints.Size; import org.springframework.http.ResponseEntity; import org.springframework.web.bind.annotation.*; import java.util.Map; @RestController @RequestMapping("/api/v1/account-deletion") public class AccountDeletionController { private final AccountDeletionService service; public AccountDeletionController( AccountDeletionService service) { this.service = service; } public record VerificationRequest( @NotBlank @Email String email, @NotBlank String provider ) {} public record ConfirmRequest( @NotBlank @Email String email, @NotBlank String provider, @NotBlank String code ) {} public record DeletionRequest( @NotBlank String verificationToken, @Size(max = 500) String reason ) {} @PostMapping("/verification") public ResponseEntity sendVerification( @Valid @RequestBody VerificationRequest request) { service.sendVerification( request.email(), request.provider()); return ResponseEntity.accepted().build(); } @PostMapping("/verification/confirm") public Map confirmVerification( @Valid @RequestBody ConfirmRequest request) { String token = service.confirmVerification( request.email(), request.provider(), request.code()); return Map.of("verificationToken", token); } @PostMapping("/requests") public Map createRequest( @Valid @RequestBody DeletionRequest request) { String requestId = service.createRequest( request.verificationToken(), request.reason()); return Map.of("requestId", requestId); } }
